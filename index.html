Pode rodar de novo. -- -- Regra do TagLine: a espessura da TÉCNICA vale para todas as cores; -- a cor só tem espessura própria quando é DIFERENTE da técnica. -- Ex.: 3D Fashion = 0,60 → 431..434 herdam 0,60; 451..454 têm 0,30 na cor. -- A vista passa a entregar a espessura que vale para cada cor: -- espessura_mm = a da cor, ou a da técnica se a cor não tiver. -- Não altera nada no TagLine. -- ===================================================================== drop view if exists estoque.vw_cartela; create view estoque.vw_cartela with (security_invoker = false) as select c.id as cor_id, c.tecnica, t.familia, t.categoria as categoria_tecnica, coalesce(t.situacao, case when t.ativa then 'em_linha' else 'fora_de_linha' end) as situacao_tecnica, c.nome as cor, c.codigo, c.pantone, c.hex, coalesce(c.espessura_mm, t.espessura_mm) as espessura_mm, -- a que vale c.espessura_mm as espessura_cor_mm, -- só se a cor for diferente t.espessura_mm as espessura_tecnica_mm, coalesce(c.situacao, case when c.ativa = false then 'fora_de_linha' else 'disponivel' end) as situacao_cor, coalesce(t.ordem, 100) as ordem_tecnica, coalesce(c.ordem, 100) as ordem_cor from desenvolvimento.tecnica_cores c join desenvolvimento.tecnicas t on t.nome = c.tecnica where estoque.est_fn_perfil_ativo(); revoke all on estoque.vw_cartela from anon; grant select on estoque.vw_cartela to authenticated; notify pgrst, 'reload schema'; -- Conferência: espessura que o estoque vai usar em cada cor (confira o 3D Fashion: 43x = 0,60 e 45x = 0,30) select c.tecnica, c.nome as cor, c.codigo, coalesce(c.espessura_mm, t.espessura_mm) as espessura_que_vale, case when c.espessura_mm is not null then 'da cor' when t.espessura_mm is not null then 'da técnica' else 'SEM ESPESSURA' end as de_onde from desenvolvimento.tecnica_cores c join desenvolvimento.tecnicas t on t.nome = c.tecnica order by c.tecnica, c.codigo nulls last, c.nome;
